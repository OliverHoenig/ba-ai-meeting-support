% Strip plot for Davison item E4: estimated share of meeting time
% spent in serious discussion (0--100 percent).
% One row per meeting group. The estimates table holds one row per participant.
% Identical estimates in a group are offset in y so both dots stay visible.
% The summary table holds the group median (mean of the two middle values)
% and the arithmetic mean. The mean is drawn just above the row so it stays
% clear of the median, which sits within one or two points of it in Groups 1 and 2.
% Compiled from the repository root via scripts/build-tikz-figures.sh.
\documentclass[11pt,border=0pt]{standalone}
\usepackage{pgfplots}
\usepackage{pgfplotstable}
\pgfplotsset{compat=1.18}
\usepackage[accentcolor=TUDa-2c]{tudacolors}
\usepackage{tudafonts}

% Same pole colors as the Likert charts.
\definecolor{likertNeg2}{RGB}{178,24,43}
\definecolor{likertPos2}{RGB}{33,102,172}

% Match the thesis typearea text width.
\newlength{\efficiencywidth}
\setlength{\efficiencywidth}{418.25555pt}

% y = 2 is the top row (Group 1). pct is the estimate in percent.
\pgfplotstableread[col sep=comma]{
  y,pct
  2,45
  2,70
  2,80
  2,100
  1,70
  1,85
  1,90
  1,98
  0,10
  0.16,60
  -0.16,60
  0,70
}\estimates

% ymean sits just above the row. Medians: 75, 87.5, 60. Means: 73.75, 85.75, 50.
\pgfplotstableread[col sep=comma]{
  y,median,mean,ymean
  2,75,73.75,2.24
  1,87.5,85.75,1.24
  0,60,50,0.24
}\summary

\begin{document}
\begin{tikzpicture}
  \begin{axis}[
      width=\efficiencywidth,
      height=4.8cm,
      xmin=0, xmax=100,
      enlarge x limits={abs=3},
      ymin=-0.55, ymax=2.55,
      ytick={0,1,2},
      yticklabels={Group 3, Group 2, Group 1},
      yticklabel style={font=\scriptsize, align=right},
      xlabel={Estimated share of meeting time (\%)},
      xlabel style={font=\scriptsize},
      xtick={0,20,40,60,80,100},
      xticklabel style={font=\scriptsize},
      xmajorgrids,
      major grid style={black!12},
      axis y line*=none,
      axis x line*=bottom,
      tick style={draw=none},
      legend cell align=left,
      legend style={
        draw=none,
        fill=none,
        at={(0.5,-0.28)},
        anchor=north,
        legend columns=-1,
        font=\scriptsize,
        /tikz/every even column/.append style={column sep=0.9em},
      },
      clip=false,
    ]
    % Within-group range, from the lowest to the highest estimate.
    \draw[black!28, line width=0.7pt]
      (axis cs:45,2) -- (axis cs:100,2)
      (axis cs:70,1) -- (axis cs:98,1)
      (axis cs:10,0) -- (axis cs:70,0);
    \addplot[
      only marks,
      mark=*,
      mark size=1.7pt,
      TUDa-1b,
    ] table[x=pct, y=y] {\estimates};
    \addplot[
      only marks,
      mark=|,
      mark size=7pt,
      line width=1.05pt,
      TUDa-9b,
    ] table[x=median, y=y] {\summary};
    \addplot[
      only marks,
      mark=diamond*,
      mark size=2.15pt,
      TUDa-8b,
    ] table[x=mean, y=ymean] {\summary};
    \legend{Individual estimate, Group median, Group mean}
    % A higher share of time in serious discussion is the favorable end.
    \node[anchor=south west, font=\scriptsize, text=likertNeg2, inner sep=0pt]
      at ([yshift=1.4ex] rel axis cs:0,1) {$\leftarrow$ Unfavorable};
    \node[anchor=south east, font=\scriptsize, text=likertPos2, inner sep=0pt]
      at ([yshift=1.4ex] rel axis cs:1,1) {Favorable $\rightarrow$};
  \end{axis}
\end{tikzpicture}
\end{document}
