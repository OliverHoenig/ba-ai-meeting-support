% Dynamic FIM reweighting (aging + cooldown). Input from the architecture chapter.
% Illustrative calibration: tau=4.3, s_hold=5, t_hold=45 s, t_cool=60 s, spoken at 12 s.
\pgfplotsset{
  IM functions/.style={
    declare function={
      tau=4.3;
      shold=5.0;
      thold=45.0;
      tcool=60.0;
      tspk=12.0;
      Hor=thold/(sqrt(1-tau/shold));
      aged(\s,\t)=max(0,\s*(1-(\t/Hor)*(\t/Hor)));
      Tlin=thold/(1-tau/shold);
      lin5(\t)=max(0,5*(1-\t/Tlin));
      lam=-(ln(tau/shold))/thold;
      exp5(\t)=5*exp(-lam*\t);
      smstep(\x)=ifthenelse(\x<0,0,ifthenelse(\x>1,1,3*\x*\x-2*\x*\x*\x));
      cool(\ts)=smstep(\ts/tcool);
      spoken(\t)=aged(5,\t)*ifthenelse(\t<tspk,1,cool(\t-tspk));
    }
  },
  IMaxis/.style={
    IM functions,
    font=\footnotesize,
    tick label style={font=\scriptsize},
    label style={font=\footnotesize},
    title style={font=\footnotesize, at={(0.0,1.0)}, anchor=south west, yshift=4pt},
    axis lines=left,
    axis line style={{}-{}},
    tick align=outside,
    clip=true,
    grid=major,
    grid style={TUDa-0a, thin},
    every axis plot/.append style={line cap=round, line join=round},
    legend cell align=left,
    legend style={
      font=\scriptsize,
      draw=none,
      fill=none,
      inner sep=1.5pt,
      row sep=-0.5pt,
      cells={anchor=west},
    },
  }
}
