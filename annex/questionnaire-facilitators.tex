\section{Semi-Structured Questionnaire for Interviews with Facilitators}

\subsection*{Background and Role}
\begin{enumerate}[label=1.\arabic*]
  \item {How do you feel about online collaboration in general?}
  \item {To start with facilitation: do you hold a generally positive or critical perspective
    toward online facilitation? Why?}
  \item {How long have you been working as a facilitator?}
  \item {Why did you start facilitating, or how did you end up becoming a facilitator?}
  \item {What do you like about being a facilitator?}
\end{enumerate}

\subsection*{Online vs. In-Person Collaboration}
\begin{enumerate}[label=2.\arabic*]
  \item {Compared with in-person collaboration, how would you rate online collaboration
    overall?}
  \item {In your experience, what are the main differences between online and in-person
    collaboration?}
\end{enumerate}

\subsection*{Facilitation Experience and Context}
\begin{enumerate}[label=3.\arabic*]
  \item {Can you describe your facilitation experience in online collaboration settings?}
  \item {Who were the participants in the sessions you facilitated?}
  \item {What were the key points you focused on when facilitating this type of event?}
  \item {What were the main difficulties or challenges you encountered as a facilitator?}
\end{enumerate}

\subsection*{Platforms and Tools}
\begin{enumerate}[label=3.\arabic*]
  \item {What platforms or tools did you use during facilitation?}
  \item {From a facilitation perspective, how did these platforms help you?}
  \item {How do you expect platforms or tools could better support you as a facilitator in the future?}
\end{enumerate}
